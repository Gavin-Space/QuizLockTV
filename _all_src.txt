17: /**
18:  * 内置局域网 HTTP 控制服务器：让同一 WiFi 下的手机/电脑用浏览器控制电视。
19:  * 路由：
20:  *   GET  /             -> 手机控制台网页(assets/console.html)
21:  *   GET  /api/status   -> 当前题库数/间隔/连对/IP 的 JSON
22:  *   POST /api/bank     -> body 为题库 JSON，校验后整库替换
23:  *   POST /api/config   -> body: interval=10&streak=3，改弹窗间隔与连对数
24:  *   POST /api/popup    -> 立即拉起全屏锁屏出题
25:  * 纯 JDK ServerSocket 实现，不依赖第三方库，targetSdk30 无 cleartext 限制问题。
26:  */
27: class ControlServer(private val ctx: Context, private val port: Int = NetworkInfo.PORT) {
28: 
29:     @Volatile private var running = false
30:     private var server: ServerSocket? = null
31: 
32:     fun start() {
33:         if (running) return
34:         running = true
35:         thread(isDaemon = true) {
36:             try {
37:                 server = ServerSocket().apply {
38:                     reuseAddress = true
39:                     bind(InetSocketAddress(port))
40:                 }
41:                 while (running) {
42:                     val socket = try { server?.accept() ?: break } catch (e: Exception) { if (!running) break else continue }
43:                     thread(isDaemon = true) { handle(socket) }
44:                 }
45:             } catch (e: Exception) {
46:                 // 端口占用等，忽略
47:             }
48:         }
49:     }
50: 
51:     fun stop() {
52:         running = false
53:         runCatching { server?.close() }
54:     }
55: 
56:     private fun handle(socket: Socket) {
57:         try {
58:             socket.use { s ->
59:                 s.soTimeout = 8000
60:                 val input = s.getInputStream()
61:                 val output = s.getOutputStream()
62:                 val reader = input.reader(StandardCharsets.UTF_8).buffered()
63: 
64:                 val requestLine = reader.readLine() ?: return
65:                 val parts = requestLine.split(" ")
66:                 if (parts.size < 2) return
67:                 val method = parts[0]
68:                 val path = parts[1]
69: 
70:                 // 读头部，拿 Content-Length
71:                 var contentLength = 0
72:                 while (true) {
73:                     val line = reader.readLine() ?: break
74:                     if (line.isEmpty()) break
75:                     if (line.lowercase().startsWith("content-length:")) {
76:                         contentLength = line.substringAfter(":").trim().toIntOrNull() ?: 0
77:                     }
78:                 }
79: 
80:                 val body = if (contentLength > 0) readNChars(reader, contentLength) else ""
81: 
82:                 route(method, path, body, output)
83:             }
84:         } catch (e: Exception) {
85:             // 单次请求出错不影响服务
86:         }
87:     }
88: 
89:     private fun readNChars(reader: BufferedReader, n: Int): String {
90:         val buf = CharArray(n)
91:         var read = 0
92:         while (read < n) {
93:             val r = reader.read(buf, read, n - read)
94:             if (r == -1) break
95:             read += r
96:         }
97:         return String(buf, 0, read)
98:     }
99: 
100:     private fun route(method: String, path: String, body: String, out: OutputStream) {
101:         when {
102:             method == "GET" && (path == "/" || path.startsWith("/index")) -> {
103:                 val html = readAsset("console.html")
104:                 send(out, 200, "text/html; charset=utf-8", html)
105:             }
106:             method == "GET" && path.startsWith("/api/status") -> {
107:                 send(out, 200, "application/json; charset=utf-8", statusJson())
108:             }
109:             method == "POST" && path.startsWith("/api/bank") -> {
110:                 handleBank(body, out)
111:             }
112:             method == "POST" && path.startsWith("/api/config") -> {
113:                 handleConfig(body, out)
114:             }
115:             method == "POST" && path.startsWith("/api/popup") -> {
116:                 handlePopup(out)
117:             }
118:             else -> send(out, 404, "text/plain; charset=utf-8", "not found")
119:         }
120:     }
121: 
122:     private fun statusJson(): String {
123:         val count = QuizStore.questions(ctx).size
124:         val interval = QuizStore.getInterval(ctx)
125:         val streak = QuizStore.getStreak(ctx)
126:         return """{"count":$count,"interval":$interval,"streak":$streak,"ip":"${NetworkInfo.localIp()}","port":$port}"""
127:     }
128: 
129:     private fun handleBank(body: String, out: OutputStream) {
130:         try {
131:             val n = QuizParser.parse(body).size
132:             QuizStore.saveRaw(ctx, body)
133:             send(out, 200, "application/json; charset=utf-8", """{"ok":true,"count":$n}""")
134:         } catch (e: Exception) {
135:             send(out, 200, "application/json; charset=utf-8", """{"ok":false,"msg":"${jsonEsc(e.message ?: "格式错误")}"}""")
136:         }
137:     }
138: 
139:     private fun handleConfig(body: String, out: OutputStream) {
140:         val map = parseForm(body)
141:         map["interval"]?.toIntOrNull()?.let { QuizStore.setInterval(ctx, it) }
142:         map["streak"]?.toIntOrNull()?.let { QuizStore.setStreak(ctx, it) }
143:         LockService.scheduleNext(ctx)
144:         send(out, 200, "application/json; charset=utf-8", statusJson())
145:     }
146: 
147:     private fun handlePopup(out: OutputStream) {
148:         val i = Intent(ctx, QuizLockActivity::class.java).addFlags(Intent.FLAG_ACTIVITY_NEW_TASK)
149:         runCatching { ctx.startActivity(i) }
150:         send(out, 200, "application/json; charset=utf-8", """{"ok":true}""")
151:     }
152: 
153:     // ---- 工具 ----
154:     private fun send(out: OutputStream, code: Int, type: String, body: String) {
155:         val bytes = body.toByteArray(StandardCharsets.UTF_8)
156:         val status = if (code == 200) "200 OK" else "$code ERR"
157:         val header = "HTTP/1.1 $status\r\n" +
158:                 "Content-Type: $type\r\n" +
159:                 "Content-Length: ${bytes.size}\r\n" +
160:                 "Cache-Control: no-store\r\n" +
161:                 "Connection: close\r\n\r\n"
162:         out.write(header.toByteArray(StandardCharsets.US_ASCII))
163:         out.write(bytes)
164:         out.flush()
165:     }
166: 
167:     private fun readAsset(name: String): String {
168:         return try {
169:             ctx.assets.open(name).use { it.readBytes().toString(StandardCharsets.UTF_8) }
170:         } catch (e: Exception) {
171:             "控制台加载失败：${e.message}"
172:         }
173:     }
174: 
175:     private fun parseForm(body: String): Map<String, String> {
176:         val m = HashMap<String, String>()
177:         body.split("&").forEach { kv ->
178:             if (kv.contains("=")) {
179:                 val k = kv.substringBefore("=")
180:                 val v = kv.substringAfter("=")
181:                 m[k] = URLDecoder.decode(v, "UTF-8")
182:             }
183:         }
184:         return m
185:     }
186: 
187:     private fun jsonEsc(s: String): String = s.replace("\\", "\\\\").replace("\"", "\\\"").replace("\n", " ").replace("\r", " ")
188: }

14: /**
15:  * 前台常驻服务：保证进程不被电视待机清理，并提供"答对解锁后重新计时 N 分钟"的闹钟调度。
16:  * 计时起点 = 上一次答对解锁的时刻，符合"每满 N 分钟弹一题"。
17:  */
18: class LockService : Service() {
19: 
20:     private var controlServer: ControlServer? = null
21: 
22:     override fun onCreate() {
23:         super.onCreate()
24:         startAsForeground()
25:         // 启动内置局域网控制服务器：同一 WiFi 下手机/电脑可远程放题与弹窗
26:         controlServer = ControlServer(applicationContext).also { it.start() }
27:     }
28: 
29:     override fun onDestroy() {
30:         super.onDestroy()
31:         runCatching { controlServer?.stop() }
32:         controlServer = null
33:     }
34: 
35:     override fun onStartCommand(intent: Intent?, flags: Int, startId: Int): Int {
36:         startAsForeground()
37:         scheduleNext(this)
38:         return START_STICKY
39:     }
40: 
41:     private fun startAsForeground() {
42:         val nm = getSystemService(Context.NOTIFICATION_SERVICE) as NotificationManager
43:         val channelId = "quiz_lock_svc"
44:         if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O) {
45:             val ch = NotificationChannel(channelId, "答题锁运行中", NotificationManager.IMPORTANCE_LOW)
46:             nm.createNotificationChannel(ch)
47:         }
48:         val builder = if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O)
49:             Notification.Builder(this, channelId) else Notification.Builder(this)
50:         val notification = builder
51:             .setContentTitle("答题锁 TV")
52:             .setContentText("运行中：答题后才能看电视")
53:             .setSmallIcon(R.drawable.ic_launcher)
54:             .build()
55:         startForeground(1, notification)
56:     }
57: 
58:     override fun onBind(intent: Intent?): IBinder? = null
59: 
60:     companion object {
61:         private const val REQ = 1001
62: 
63:         /** 安排 interval 分钟后弹出下一题。 */
64:         fun scheduleNext(ctx: Context) {
65:             val am = ctx.getSystemService(Context.ALARM_SERVICE) as AlarmManager
66:             val pi = pending(ctx)
67:             val intervalMin = QuizStore.getInterval(ctx)
68:             val triggerAt = System.currentTimeMillis() + intervalMin * 60_000L
69:             try {
70:                 if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.M) {
71:                     am.setExactAndAllowWhileIdle(AlarmManager.RTC_WAKEUP, triggerAt, pi)
72:                 } else {
73:                     am.setExact(AlarmManager.RTC_WAKEUP, triggerAt, pi)
74:                 }
75:             } catch (e: SecurityException) {
76:                 am.set(AlarmManager.RTC_WAKEUP, triggerAt, pi)
77:             }
78:         }
79: 
80:         private fun pending(ctx: Context): PendingIntent {
81:             val i = Intent(ctx, AlarmReceiver::class.java)
82:             val flag = if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.M)
83:                 PendingIntent.FLAG_IMMUTABLE or PendingIntent.FLAG_UPDATE_CURRENT
84:             else PendingIntent.FLAG_UPDATE_CURRENT
85:             return PendingIntent.getBroadcast(ctx, REQ, i, flag)
86:         }
87:     }
88: }

### Question.kt
1: package com.huiyang.quizlocktv
2: 
3: import org.json.JSONArray
4: import org.json.JSONObject
5: 
6: /**
7:  * 一道题。
8:  * type = "single"(单选) 或 "judge"(判断)
9:  * correct: single=正确选项下标(0基)；judge=0表示错误/false, 1表示正确/true
10:  */
11: data class Question(
12:     val type: String,
13:     val stem: String,
14:     val options: List<String>,   // 判断题为空列表
15:     val correct: Int,
16:     val analysis: String
17: ) {
18:     val isJudge get() = type == "judge"
19: }
20: 
21: object QuizParser {
22:     /** 解析题库 JSON，返回题目列表；格式错误时抛 IllegalArgumentException */
23:     fun parse(text: String): List<Question> {
24:         val root = JSONObject(text.trim())
25:         val arr = root.optJSONArray("questions")
26:             ?: throw IllegalArgumentException("缺少 questions 数组")
27:         val list = ArrayList<Question>()
28:         for (i in 0 until arr.length()) {
29:             val o = arr.getJSONObject(i)
30:             val type = o.optString("type")
31:             val stem = o.optString("stem")
32:             if (stem.isBlank()) continue
33:             when (type) {
34:                 "single" -> {
35:                     val opts = ArrayList<String>()
36:                     val oa = o.optJSONArray("options") ?: JSONArray()
37:                     for (k in 0 until oa.length()) opts.add(oa.getString(k))
38:                     if (opts.isEmpty()) continue
39:                     val ans = o.optInt("answer", -1)
40:                     if (ans < 0 || ans >= opts.size) continue
41:                     list.add(Question(type, stem, opts, ans, o.optString("analysis")))
42:                 }
43:                 "judge" -> {
44:                     val ans = if (o.optBoolean("answer", false)) 1 else 0
45:                     list.add(Question(type, stem, emptyList(), ans, o.optString("analysis")))
46:                 }
47:                 else -> { /* 跳过未知题型 */ }
48:             }
49:         }
50:         if (list.isEmpty()) throw IllegalArgumentException("没有解析到任何有效题目，请检查 JSON 格式")
51:         return list
52:     }
53: }

### QuizStore.kt
1: package com.huiyang.quizlocktv
2: 
3: import android.content.Context
4: import android.content.SharedPreferences
5: 
6: /**
7:  * 用 SharedPreferences 保存：题库原文、当前进度指针、出题间隔、同步地址、上次同步时间。
8:  * 题库以 JSON 字符串整体存储，解析结果缓存到内存。
9:  */
10: object QuizStore {
11:     private const val PREF = "quiz_lock_pref"
12:     private const val KEY_RAW = "quiz_raw_json"        // 题库原文 JSON
13:     private const val KEY_POINTER = "pointer"          // 当前出到第几题（循环推进）
14:     private const val KEY_INTERVAL = "interval_min"    // 弹窗间隔(分钟)
15:     private const val KEY_URL = "sync_url"             // 入口B：自动同步地址
16:     private const val KEY_LAST_SYNC = "last_sync"      // 上次同步成功时间戳
17:     private const val KEY_STREAK = "required_streak"   // 需连续答对几题才解锁（防乱点）
18: 
19:     private var cache: List<Question>? = null
20: 
21:     private fun sp(c: Context): SharedPreferences =
22:         c.getSharedPreferences(PREF, Context.MODE_PRIVATE)
23: 
24:     // ---- 题库读写 ----
25:     fun saveRaw(c: Context, rawJson: String) {
26:         sp(c).edit().putString(KEY_RAW, rawJson).putInt(KEY_POINTER, 0).apply()
27:         cache = null // 题库更新，重置缓存与进度
28:     }
29: 
30:     fun currentRaw(c: Context): String = sp(c).getString(KEY_RAW, "") ?: ""
31: 
32:     fun questions(c: Context): List<Question> {
33:         cache?.let { return it }
34:         val raw = currentRaw(c)
35:         val parsed = if (raw.isBlank()) emptyList() else
36:             try { QuizParser.parse(raw) } catch (e: Exception) { emptyList() }
37:         cache = parsed
38:         return parsed
39:     }
40: 
41:     fun hasBank(c: Context): Boolean = questions(c).isNotEmpty()
42: 
43:     /**
44:      * 取下一道未出过的题：返回当前指针下标，并把指针循环推进到下一题。
45:      * 首次弹出和答错后追加新题都调用本方法，保证每次都是不同的题、不重复。
46:      */
47:     fun takeNext(c: Context): Int {
48:         val n = questions(c).size
49:         if (n == 0) return -1
50:         val p = sp(c).getInt(KEY_POINTER, 0) % n
51:         sp(c).edit().putInt(KEY_POINTER, (p + 1) % n).apply()
52:         return p
53:     }
54: 
55:     // ---- 配置 ----
56:     fun setInterval(c: Context, min: Int) {
57:         sp(c).edit().putInt(KEY_INTERVAL, min.coerceIn(1, 1440)).apply()
58:     }
59:     fun getInterval(c: Context): Int = sp(c).getInt(KEY_INTERVAL, 10)
60: 
61:     fun setSyncUrl(c: Context, url: String) {
62:         sp(c).edit().putString(KEY_URL, url.trim()).apply()
63:     }
64:     fun getSyncUrl(c: Context): String = sp(c).getString(KEY_URL, "") ?: ""
65: 
66:     fun setLastSync(c: Context, t: Long) {
67:         sp(c).edit().putLong(KEY_LAST_SYNC, t).apply()
68:     }
69:     fun getLastSync(c: Context): Long = sp(c).getLong(KEY_LAST_SYNC, 0L)
70: 
71:     /** 解锁所需连续答对题数，默认 3，可设 1~10。设越大越难靠蒙混过关。 */
72:     fun setStreak(c: Context, n: Int) {
73:         sp(c).edit().putInt(KEY_STREAK, n.coerceIn(1, 10)).apply()
74:     }
75:     fun getStreak(c: Context): Int = sp(c).getInt(KEY_STREAK, 3)
76: }
### QuizSync.kt
1: package com.huiyang.quizlocktv
2: 
3: import android.content.Context
4: import android.os.Handler
5: import android.os.Looper
6: import java.net.HttpURLConnection
7: import java.net.URL
8: 
9: /**
10:  * 入口B：从用户填的 URL 拉取题库 JSON。fetchNow 内部自起后台线程，回调在主线程。
11:  */
12: object QuizSync {
13:     interface Result {
14:         fun onDone(ok: Boolean, msg: String)
15:     }
16: 
17:     /** 同步：拉取 URL，成功且能解析则替换本地题库。 */
18:     fun fetchNow(ctx: Context, result: Result?) {
19:         val url = QuizStore.getSyncUrl(ctx)
20:         Thread {
21:             if (url.isBlank()) {
22:                 post(result, false, "未设置同步地址")
23:                 return@Thread
24:             }
25:             try {
26:                 val text = httpGet(url)
27:                 if (text == null) {
28:                     post(result, false, "无法访问地址（检查网络/链接是否公开可读）")
29:                     return@Thread
30:                 }
31:                 // 先验证能解析成题库，再入库
32:                 QuizParser.parse(text)
33:                 QuizStore.saveRaw(ctx, text)
34:                 QuizStore.setLastSync(ctx, System.currentTimeMillis())
35:                 post(result, true, "同步成功，题库已更新")
36:             } catch (e: Exception) {
37:                 post(result, false, "同步失败：${e.message}")
38:             }
39:         }.start()
40:     }
41: 
42:     private fun httpGet(u: String): String? {
43:         var conn: HttpURLConnection? = null
44:         return try {
45:             conn = (URL(u).openConnection() as HttpURLConnection).apply {
46:                 connectTimeout = 10000
47:                 readTimeout = 10000
48:                 requestMethod = "GET"
49:                 instanceFollowRedirects = true
50:             }
51:             if (conn.responseCode in 200..299) {
52:                 conn.inputStream.bufferedReader().use { it.readText() }
53:             } else null
54:         } catch (e: Exception) {
55:             null
56:         } finally {
57:             conn?.disconnect()
58:         }
59:     }
60: 
61:     private fun post(r: Result?, ok: Boolean, msg: String) {
62:         Handler(Looper.getMainLooper()).post { r?.onDone(ok, msg) }
63:     }
64: }
