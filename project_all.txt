package com.huiyang.quizlocktv

import android.content.BroadcastReceiver
import android.content.Context
import android.content.Intent
import android.os.Build

/**
 * 开机完成：启动前台服务（负责安排每 N 分钟弹窗），并立即弹"开机首题"。
 */
class BootReceiver : BroadcastReceiver() {
    override fun onReceive(ctx: Context, intent: Intent) {
        val svc = Intent(ctx, LockService::class.java)
        try {
            if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O) {
                ctx.startForegroundService(svc)
            } else {
                ctx.startService(svc)
            }
        } catch (e: Exception) { /* 个别 ROM 限制，忽略 */ }

        // 开机先做一道题：从后台拉起锁屏页
        val lock = Intent(ctx, QuizLockActivity::class.java).apply {
            addFlags(Intent.FLAG_ACTIVITY_NEW_TASK)
            putExtra(QuizLockActivity.EXTRA_FIRST, true)
        }
        try { ctx.startActivity(lock) } catch (e: Exception) {}
    }
}

package com.huiyang.quizlocktv

import android.content.Context
import android.content.Intent
import java.io.BufferedReader
import java.io.ByteArrayOutputStream
import java.io.InputStream
import java.io.OutputStream
import java.net.InetSocketAddress
import java.net.NetworkInterface
import java.net.ServerSocket
import java.net.Socket
import java.net.URLDecoder
import java.nio.charset.StandardCharsets
import kotlin.concurrent.thread

/**
 * 内置局域网 HTTP 控制服务器：让同一 WiFi 下的手机/电脑用浏览器控制电视。
 * 路由：
 *   GET  /             -> 手机控制台网页(assets/console.html)
 *   GET  /api/status   -> 当前题库数/间隔/连对/IP 的 JSON
 *   POST /api/bank     -> body 为题库 JSON，校验后整库替换
 *   POST /api/config   -> body: interval=10&streak=3，改弹窗间隔与连对数
 *   POST /api/popup    -> 立即拉起全屏锁屏出题
 * 纯 JDK ServerSocket 实现，不依赖第三方库，targetSdk30 无 cleartext 限制问题。
 */
class ControlServer(private val ctx: Context, private val port: Int = NetworkInfo.PORT) {

    @Volatile private var running = false
    private var server: ServerSocket? = null

    fun start() {
        if (running) return
        running = true
        thread(isDaemon = true) {
            try {
                server = ServerSocket().apply {
                    reuseAddress = true
                    bind(InetSocketAddress(port))
                }
                while (running) {
                    val socket = try { server?.accept() ?: break } catch (e: Exception) { if (!running) break else continue }
                    thread(isDaemon = true) { handle(socket) }
                }
            } catch (e: Exception) {
                // 端口占用等，忽略
            }
        }
    }

    fun stop() {
        running = false
        runCatching { server?.close() }
    }

    private fun handle(socket: Socket) {
        try {
            socket.use { s ->
                s.soTimeout = 8000
                val input = s.getInputStream()
                val output = s.getOutputStream()
                val reader = input.reader(StandardCharsets.UTF_8).buffered()

                val requestLine = reader.readLine() ?: return
                val parts = requestLine.split(" ")
                if (parts.size < 2) return
                val method = parts[0]
                val path = parts[1]

                // 读头部，拿 Content-Length
                var contentLength = 0
                while (true) {
                    val line = reader.readLine() ?: break
                    if (line.isEmpty()) break
                    if (line.lowercase().startsWith("content-length:")) {
                        contentLength = line.substringAfter(":").trim().toIntOrNull() ?: 0
                    }
                }

                val body = if (contentLength > 0) readNChars(reader, contentLength) else ""

                route(method, path, body, output)
            }
        } catch (e: Exception) {
            // 单次请求出错不影响服务
        }
    }

    private fun readNChars(reader: BufferedReader, n: Int): String {
        val buf = CharArray(n)
        var read = 0
        while (read < n) {
            val r = reader.read(buf, read, n - read)
            if (r == -1) break
            read += r
        }
        return String(buf, 0, read)
    }

    private fun route(method: String, path: String, body: String, out: OutputStream) {
        when {
            method == "GET" && (path == "/" || path.startsWith("/index")) -> {
                val html = readAsset("console.html")
                send(out, 200, "text/html; charset=utf-8", html)
            }
            method == "GET" && path.startsWith("/api/status") -> {
                send(out, 200, "application/json; charset=utf-8", statusJson())
            }
            method == "POST" && path.startsWith("/api/bank") -> {
                handleBank(body, out)
            }
            method == "POST" && path.startsWith("/api/config") -> {
                handleConfig(body, out)
            }
            method == "POST" && path.startsWith("/api/popup") -> {
                handlePopup(out)
            }
            else -> send(out, 404, "text/plain; charset=utf-8", "not found")
        }
    }

    private fun statusJson(): String {
        val count = QuizStore.questions(ctx).size
        val interval = QuizStore.getInterval(ctx)
        val streak = QuizStore.getStreak(ctx)
        return """{"count":$count,"interval":$interval,"streak":$streak,"ip":"${NetworkInfo.localIp()}","port":$port}"""
    }

    private fun handleBank(body: String, out: OutputStream) {
        try {
            val n = QuizParser.parse(body).size
            QuizStore.saveRaw(ctx, body)
            send(out, 200, "application/json; charset=utf-8", """{"ok":true,"count":$n}""")
        } catch (e: Exception) {
            send(out, 200, "application/json; charset=utf-8", """{"ok":false,"msg":"${jsonEsc(e.message ?: "格式错误")}"}""")
        }
    }

    private fun handleConfig(body: String, out: OutputStream) {
        val map = parseForm(body)
        map["interval"]?.toIntOrNull()?.let { QuizStore.setInterval(ctx, it) }
        map["streak"]?.toIntOrNull()?.let { QuizStore.setStreak(ctx, it) }
        LockService.scheduleNext(ctx)
        send(out, 200, "application/json; charset=utf-8", statusJson())
    }

    private fun handlePopup(out: OutputStream) {
        val i = Intent(ctx, QuizLockActivity::class.java).addFlags(Intent.FLAG_ACTIVITY_NEW_TASK)
        runCatching { ctx.startActivity(i) }
        send(out, 200, "application/json; charset=utf-8", """{"ok":true}""")
    }

    // ---- 工具 ----
    private fun send(out: OutputStream, code: Int, type: String, body: String) {
        val bytes = body.toByteArray(StandardCharsets.UTF_8)
        val status = if (code == 200) "200 OK" else "$code ERR"
        val header = "HTTP/1.1 $status\r\n" +
                "Content-Type: $type\r\n" +
                "Content-Length: ${bytes.size}\r\n" +
                "Cache-Control: no-store\r\n" +
                "Connection: close\r\n\r\n"
        out.write(header.toByteArray(StandardCharsets.US_ASCII))
        out.write(bytes)
        out.flush()
    }

    private fun readAsset(name: String): String {
        return try {
            ctx.assets.open(name).use { it.readBytes().toString(StandardCharsets.UTF_8) }
        } catch (e: Exception) {
            "控制台加载失败：${e.message}"
        }
    }

    private fun parseForm(body: String): Map<String, String> {
        val m = HashMap<String, String>()
        body.split("&").forEach { kv ->
            if (kv.contains("=")) {
                val k = kv.substringBefore("=")
                val v = kv.substringAfter("=")
                m[k] = URLDecoder.decode(v, "UTF-8")
            }
        }
        return m
    }

    private fun jsonEsc(s: String): String = s.replace("\\", "\\\\").replace("\"", "\\\"").replace("\n", " ").replace("\r", " ")
}

package com.huiyang.quizlocktv

import android.app.AlarmManager
import android.app.Notification
import android.app.NotificationChannel
import android.app.NotificationManager
import android.app.PendingIntent
import android.app.Service
import android.content.Context
import android.content.Intent
import android.os.Build
import android.os.IBinder

/**
 * 前台常驻服务：保证进程不被电视待机清理，并提供"答对解锁后重新计时 N 分钟"的闹钟调度。
 * 计时起点 = 上一次答对解锁的时刻，符合"每满 N 分钟弹一题"。
 */
class LockService : Service() {

    private var controlServer: ControlServer? = null

    override fun onCreate() {
        super.onCreate()
        startAsForeground()
        // 启动内置局域网控制服务器：同一 WiFi 下手机/电脑可远程放题与弹窗
        controlServer = ControlServer(applicationContext).also { it.start() }
    }

    override fun onDestroy() {
        super.onDestroy()
        runCatching { controlServer?.stop() }
        controlServer = null
    }

    override fun onStartCommand(intent: Intent?, flags: Int, startId: Int): Int {
        startAsForeground()
        scheduleNext(this)
        return START_STICKY
    }

    private fun startAsForeground() {
        val nm = getSystemService(Context.NOTIFICATION_SERVICE) as NotificationManager
        val channelId = "quiz_lock_svc"
        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O) {
            val ch = NotificationChannel(channelId, "答题锁运行中", NotificationManager.IMPORTANCE_LOW)
            nm.createNotificationChannel(ch)
        }
        val builder = if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O)
            Notification.Builder(this, channelId) else Notification.Builder(this)
        val notification = builder
            .setContentTitle("答题锁 TV")
            .setContentText("运行中：答题后才能看电视")
            .setSmallIcon(R.drawable.ic_launcher)
            .build()
        startForeground(1, notification)
    }

    override fun onBind(intent: Intent?): IBinder? = null

    companion object {
        private const val REQ = 1001

        /** 安排 interval 分钟后弹出下一题。 */
        fun scheduleNext(ctx: Context) {
            val am = ctx.getSystemService(Context.ALARM_SERVICE) as AlarmManager
            val pi = pending(ctx)
            val intervalMin = QuizStore.getInterval(ctx)
            val triggerAt = System.currentTimeMillis() + intervalMin * 60_000L
            try {
                if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.M) {
                    am.setExactAndAllowWhileIdle(AlarmManager.RTC_WAKEUP, triggerAt, pi)
                } else {
                    am.setExact(AlarmManager.RTC_WAKEUP, triggerAt, pi)
                }
            } catch (e: SecurityException) {
                am.set(AlarmManager.RTC_WAKEUP, triggerAt, pi)
            }
        }

        private fun pending(ctx: Context): PendingIntent {
            val i = Intent(ctx, AlarmReceiver::class.java)
            val flag = if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.M)
                PendingIntent.FLAG_IMMUTABLE or PendingIntent.FLAG_UPDATE_CURRENT
            else PendingIntent.FLAG_UPDATE_CURRENT
            return PendingIntent.getBroadcast(ctx, REQ, i, flag)
        }
    }
}

package com.huiyang.quizlocktv

import java.net.NetworkInterface

/** 获取电视在局域网里的 IPv4 地址（手机据此访问控制台）。 */
object NetworkInfo {
    const val PORT = 8770

    fun localIp(): String {
        try {
            val en = NetworkInterface.getNetworkInterfaces()
            for (intf in en) {
                if (!intf.isUp || intf.isLoopback) continue
                for (addr in intf.inetAddresses) {
                    if (!addr.isLoopbackAddress && addr.hostAddress?.contains(":") != true) {
                        return addr.hostAddress ?: ""
                    }
                }
            }
        } catch (e: Exception) {}
        return "未知"
    }

    /** 手机浏览器要访问的完整地址，如 http://192.168.1.50:8770 */
    fun consoleUrl(): String = "http://${localIp()}:$PORT"
}

package com.huiyang.quizlocktv

import org.json.JSONArray
import org.json.JSONObject

/**
 * 一道题。
 * type = "single"(单选) 或 "judge"(判断)
 * correct: single=正确选项下标(0基)；judge=0表示错误/false, 1表示正确/true
 */
data class Question(
    val type: String,
    val stem: String,
    val options: List<String>,   // 判断题为空列表
    val correct: Int,
    val analysis: String
) {
    val isJudge get() = type == "judge"
}

object QuizParser {
    /** 解析题库 JSON，返回题目列表；格式错误时抛 IllegalArgumentException */
    fun parse(text: String): List<Question> {
        val root = JSONObject(text.trim())
        val arr = root.optJSONArray("questions")
            ?: throw IllegalArgumentException("缺少 questions 数组")
        val list = ArrayList<Question>()
        for (i in 0 until arr.length()) {
            val o = arr.getJSONObject(i)
            val type = o.optString("type")
            val stem = o.optString("stem")
            if (stem.isBlank()) continue
            when (type) {
                "single" -> {
                    val opts = ArrayList<String>()
                    val oa = o.optJSONArray("options") ?: JSONArray()
                    for (k in 0 until oa.length()) opts.add(oa.getString(k))
                    if (opts.isEmpty()) continue
                    val ans = o.optInt("answer", -1)
                    if (ans < 0 || ans >= opts.size) continue
                    list.add(Question(type, stem, opts, ans, o.optString("analysis")))
                }
                "judge" -> {
                    val ans = if (o.optBoolean("answer", false)) 1 else 0
                    list.add(Question(type, stem, emptyList(), ans, o.optString("analysis")))
                }
                else -> { /* 跳过未知题型 */ }
            }
        }
        if (list.isEmpty()) throw IllegalArgumentException("没有解析到任何有效题目，请检查 JSON 格式")
        return list
    }
}

import android.app.Activity
import android.graphics.Color
import android.graphics.Typeface

/**
 * 全屏锁屏出题页：遮住电视画面，必须连续答对才放行。
 * 防乱点两道防线：
 *  1) 需连续答对 requiredStreak 题才解锁，答错则计数清零重来；
 *  2) 单选题选项每次随机打乱，杜绝"闭眼点第一个/固定A"。
 * 仍保持规则：每次一题、答错追加一道新题直到答对、答对解锁继续看。
 */
class QuizLockActivity : Activity() {

    companion object {
        const val EXTRA_FIRST = "first"
    }

    private lateinit var stemView: TextView
    private lateinit var feedbackView: TextView
    private lateinit var progressView: TextView
    private lateinit var optionsBox: LinearLayout

    private var requiredStreak = 3
    private var correctCount = 0
    private var current: Question? = null

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        window.addFlags(
            WindowManager.LayoutParams.FLAG_KEEP_SCREEN_ON or
            WindowManager.LayoutParams.FLAG_SECURE or
            WindowManager.LayoutParams.FLAG_SHOW_WHEN_LOCKED or
            WindowManager.LayoutParams.FLAG_TURN_SCREEN_ON or
            WindowManager.LayoutParams.FLAG_DISMISS_KEYGUARD
        )
        requiredStreak = QuizStore.getStreak(this)
        correctCount = 0
        buildLayout()
        showNextQuestion()
    }

    private fun buildLayout() {
        val root = ScrollView(this).apply {
            setBackgroundColor(Color.parseColor("#0B1220"))
            isFillViewport = true
        }
        val col = LinearLayout(this).apply {
            orientation = LinearLayout.VERTICAL
            setPadding(dp(28), dp(28), dp(28), dp(28))
            gravity = Gravity.CENTER_HORIZONTAL
        }

        val title = TextView(this).apply {
            text = "🔒 答题解锁看电视（需连对 ${requiredStreak} 题）"
            textSize = 22f
            setTypeface(typeface, Typeface.BOLD)
            setTextColor(Color.parseColor("#2BB673"))
            gravity = Gravity.CENTER
        }

        progressView = TextView(this).apply {
            textSize = 18f
            setTextColor(Color.parseColor("#9FB0CC"))
            gravity = Gravity.CENTER
            setPadding(0, dp(6), 0, dp(4))
        }

        stemView = TextView(this).apply {
            textSize = 26f
            setTypeface(typeface, Typeface.BOLD)
            setTextColor(Color.WHITE)
            setPadding(0, dp(16), 0, dp(16))
        }

        optionsBox = LinearLayout(this).apply { orientation = LinearLayout.VERTICAL }

        feedbackView = TextView(this).apply {
            textSize = 18f
            setTextColor(Color.parseColor("#FF5B5B"))
            setPadding(0, dp(16), 0, dp(8))
        }

        col.addView(title)
        col.addView(progressView)
        col.addView(stemView)
        col.addView(optionsBox)
        col.addView(feedbackView)
        root.addView(col)
        setContentView(root)
        updateProgress()
    }

    private fun updateProgress() {
        progressView.text = "已连续答对 $correctCount / $requiredStreak 题"
    }

    private fun showNextQuestion() {
        val qs = QuizStore.questions(this)
        if (qs.isEmpty()) {
            Toast.makeText(this, "题库为空，请先导入题库", Toast.LENGTH_LONG).show()
            finish()
            return
        }
        val idx = QuizStore.takeNext(this)
        if (idx < 0 || idx >= qs.size) { finish(); return }
        val q = qs[idx]
        current = q
        render(q)
    }

    private fun render(q: Question) {
        stemView.text = if (q.isJudge) "【判断题】${q.stem}" else "【单选题】${q.stem}"
        feedbackView.text = ""
        updateProgress()
        optionsBox.removeAllViews()

        if (q.isJudge) {
            // q.correct==1 表示答案为“正确”，==0 表示答案为“错误”
            optionsBox.addView(makeOptionButton("正确", q.correct == 1, q))
            optionsBox.addView(makeOptionButton("错误", q.correct == 0, q))
        } else {
            // 选项连内容一起随机打乱，正确答案跟随其文本移动
            val items = ArrayList<Pair<String, Int>>()
            for (i in q.options.indices) items.add(q.options[i] to i) // (文本, 原下标)
            items.shuffle(Random.Default)
            items.forEachIndexed { pos, pair ->
                val letter = ('A' + pos)
                val isCorrect = (pair.second == q.correct)
                optionsBox.addView(
                    makeOptionButton("$letter. ${pair.first}", isCorrect, q)
                )
            }
        }
    }

    private fun makeOptionButton(label: String, isCorrect: Boolean, q: Question): Button {
        val b = Button(this).apply {
            text = label
            textSize = 20f
            isAllCaps = false
            setPadding(dp(16), dp(14), dp(16), dp(14))
            layoutParams = LinearLayout.LayoutParams(
                LinearLayout.LayoutParams.MATCH_PARENT,
                LinearLayout.LayoutParams.WRAP_CONTENT
            ).apply { bottomMargin = dp(10) }
            setBackgroundColor(Color.parseColor("#16243B"))
            setTextColor(Color.WHITE)
        }
        b.setOnClickListener {
            if (isCorrect) {
                correctCount++
                updateProgress()
                if (correctCount >= requiredStreak) {
                    // 达成连对数：放行 + 重新计时
                    feedbackView.setTextColor(Color.parseColor("#2BB673"))
                    feedbackView.text = "✅ 连对 ${requiredStreak} 题，继续看吧！"
                    LockService.scheduleNext(this)
                    Handler(Looper.getMainLooper()).postDelayed({ finish() }, 800)
                } else {
                    // 还需继续连对：追加下一道新题
                    feedbackView.setTextColor(Color.parseColor("#2BB673"))
                    feedbackView.text = "✅ 答对了！再答对 ${requiredStreak - correctCount} 题即可解锁"
                    showNextQuestion()
                }
            } else {
                // 答错：连对清零，追加一道新题，仍锁屏
                correctCount = 0
                updateProgress()
                feedbackView.setTextColor(Color.parseColor("#FF5B5B"))
                val tip = if (q.analysis.isNotBlank())
                    "❌ 答错了，计数清零。${q.analysis}  再来一题"
                else "❌ 答错了，计数清零，再来一题"
                feedbackView.text = tip
                showNextQuestion()
            }
        }
        return b
    }

    override fun onKeyDown(keyCode: Int, event: KeyEvent): Boolean {
        if (keyCode == KeyEvent.KEYCODE_BACK || keyCode == KeyEvent.KEYCODE_HOME) return true
        return super.onKeyDown(keyCode, event)
    }

    override fun onBackPressed() { /* 禁用 */ }

    private fun dp(v: Int): Int = (v * resources.displayMetrics.density).toInt()
}

package com.huiyang.quizlocktv

import android.content.Context
import android.content.SharedPreferences

/**
 * 用 SharedPreferences 保存：题库原文、当前进度指针、出题间隔、同步地址、上次同步时间。
 * 题库以 JSON 字符串整体存储，解析结果缓存到内存。
 */
object QuizStore {
    private const val PREF = "quiz_lock_pref"
    private const val KEY_RAW = "quiz_raw_json"        // 题库原文 JSON
    private const val KEY_POINTER = "pointer"          // 当前出到第几题（循环推进）
    private const val KEY_INTERVAL = "interval_min"    // 弹窗间隔(分钟)
    private const val KEY_URL = "sync_url"             // 入口B：自动同步地址
    private const val KEY_LAST_SYNC = "last_sync"      // 上次同步成功时间戳
    private const val KEY_STREAK = "required_streak"   // 需连续答对几题才解锁（防乱点）

    private var cache: List<Question>? = null

    private fun sp(c: Context): SharedPreferences =
        c.getSharedPreferences(PREF, Context.MODE_PRIVATE)

    // ---- 题库读写 ----
    fun saveRaw(c: Context, rawJson: String) {
        sp(c).edit().putString(KEY_RAW, rawJson).putInt(KEY_POINTER, 0).apply()
        cache = null // 题库更新，重置缓存与进度
    }

    fun currentRaw(c: Context): String = sp(c).getString(KEY_RAW, "") ?: ""

    fun questions(c: Context): List<Question> {
        cache?.let { return it }
        val raw = currentRaw(c)
        val parsed = if (raw.isBlank()) emptyList() else
            try { QuizParser.parse(raw) } catch (e: Exception) { emptyList() }
        cache = parsed
        return parsed
    }

    fun hasBank(c: Context): Boolean = questions(c).isNotEmpty()

    /**
     * 取下一道未出过的题：返回当前指针下标，并把指针循环推进到下一题。
     * 首次弹出和答错后追加新题都调用本方法，保证每次都是不同的题、不重复。
     */
    fun takeNext(c: Context): Int {
        val n = questions(c).size
        if (n == 0) return -1
        val p = sp(c).getInt(KEY_POINTER, 0) % n
        sp(c).edit().putInt(KEY_POINTER, (p + 1) % n).apply()
        return p
    }

    // ---- 配置 ----
    fun setInterval(c: Context, min: Int) {
        sp(c).edit().putInt(KEY_INTERVAL, min.coerceIn(1, 1440)).apply()
    }
    fun getInterval(c: Context): Int = sp(c).getInt(KEY_INTERVAL, 10)

    fun setSyncUrl(c: Context, url: String) {
        sp(c).edit().putString(KEY_URL, url.trim()).apply()
    }
    fun getSyncUrl(c: Context): String = sp(c).getString(KEY_URL, "") ?: ""

    fun setLastSync(c: Context, t: Long) {
        sp(c).edit().putLong(KEY_LAST_SYNC, t).apply()
    }
    fun getLastSync(c: Context): Long = sp(c).getLong(KEY_LAST_SYNC, 0L)

    /** 解锁所需连续答对题数，默认 3，可设 1~10。设越大越难靠蒙混过关。 */
    fun setStreak(c: Context, n: Int) {
        sp(c).edit().putInt(KEY_STREAK, n.coerceIn(1, 10)).apply()
    }
    fun getStreak(c: Context): Int = sp(c).getInt(KEY_STREAK, 3)
}

package com.huiyang.quizlocktv

import android.content.Context
import android.os.Handler
import android.os.Looper
import java.net.HttpURLConnection
import java.net.URL

/**
 * 入口B：从用户填的 URL 拉取题库 JSON。fetchNow 内部自起后台线程，回调在主线程。
 */
object QuizSync {
    interface Result {
        fun onDone(ok: Boolean, msg: String)
    }

    /** 同步：拉取 URL，成功且能解析则替换本地题库。 */
    fun fetchNow(ctx: Context, result: Result?) {
        val url = QuizStore.getSyncUrl(ctx)
        Thread {
            if (url.isBlank()) {
                post(result, false, "未设置同步地址")
                return@Thread
            }
            try {
                val text = httpGet(url)
                if (text == null) {
                    post(result, false, "无法访问地址（检查网络/链接是否公开可读）")
                    return@Thread
                }
                // 先验证能解析成题库，再入库
                QuizParser.parse(text)
                QuizStore.saveRaw(ctx, text)
                QuizStore.setLastSync(ctx, System.currentTimeMillis())
                post(result, true, "同步成功，题库已更新")
            } catch (e: Exception) {
                post(result, false, "同步失败：${e.message}")
            }
        }.start()
    }

    private fun httpGet(u: String): String? {
        var conn: HttpURLConnection? = null
        return try {
            conn = (URL(u).openConnection() as HttpURLConnection).apply {
                connectTimeout = 10000
                readTimeout = 10000
                requestMethod = "GET"
                instanceFollowRedirects = true
            }
            if (conn.responseCode in 200..299) {
                conn.inputStream.bufferedReader().use { it.readText() }
            } else null
        } catch (e: Exception) {
            null
        } finally {
            conn?.disconnect()
        }
    }

    private fun post(r: Result?, ok: Boolean, msg: String) {
        Handler(Looper.getMainLooper()).post { r?.onDone(ok, msg) }
    }
}

package com.huiyang.quizlocktv

import android.app.Activity
import android.graphics.Color
import android.graphics.Typeface
import android.os.Bundle
import android.os.Handler
import android.os.Looper
import android.view.Gravity
import android.view.KeyEvent
import android.view.WindowManager
import android.widget.Button
import android.widget.LinearLayout
import android.widget.ScrollView
import android.widget.TextView
import android.widget.Toast
import kotlin.random.Random

/**
 * 全屏锁屏出题页：遮住电视画面，必须连续答对才放行。
 * 防乱点两道防线：
 *  1) 需连续答对 requiredStreak 题才解锁，答错则计数清零重来；
 *  2) 单选题选项每次随机打乱，杜绝"闭眼点第一个/固定A"。
 * 仍保持规则：每次一题、答错追加一道新题直到答对、答对解锁继续看。
 */
class QuizLockActivity : Activity() {

    companion object {
        const val EXTRA_FIRST = "first"
    }

    private lateinit var stemView: TextView
    private lateinit var feedbackView: TextView
    private lateinit var progressView: TextView
    private lateinit var optionsBox: LinearLayout

    private var requiredStreak = 3
    private var correctCount = 0
    private var current: Question? = null

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        window.addFlags(
            WindowManager.LayoutParams.FLAG_KEEP_SCREEN_ON or
            WindowManager.LayoutParams.FLAG_SECURE or
            WindowManager.LayoutParams.FLAG_SHOW_WHEN_LOCKED or
            WindowManager.LayoutParams.FLAG_TURN_SCREEN_ON or
            WindowManager.LayoutParams.FLAG_DISMISS_KEYGUARD
        )
        requiredStreak = QuizStore.getStreak(this)
        correctCount = 0
        buildLayout()
        showNextQuestion()
    }

    private fun buildLayout() {
        val root = ScrollView(this).apply {
            setBackgroundColor(Color.parseColor("#0B1220"))
            isFillViewport = true
        }
        val col = LinearLayout(this).apply {
            orientation = LinearLayout.VERTICAL
            setPadding(dp(28), dp(28), dp(28), dp(28))
            gravity = Gravity.CENTER_HORIZONTAL
        }

        val title = TextView(this).apply {
            text = "🔒 答题解锁看电视（需连对 ${requiredStreak} 题）"
            textSize = 22f
            setTypeface(typeface, Typeface.BOLD)
            setTextColor(Color.parseColor("#2BB673"))
            gravity = Gravity.CENTER
        }

        progressView = TextView(this).apply {
            textSize = 18f
            setTextColor(Color.parseColor("#9FB0CC"))
            gravity = Gravity.CENTER
            setPadding(0, dp(6), 0, dp(4))
        }

        stemView = TextView(this).apply {
            textSize = 26f
            setTypeface(typeface, Typeface.BOLD)
            setTextColor(Color.WHITE)
            setPadding(0, dp(16), 0, dp(16))
        }

        optionsBox = LinearLayout(this).apply { orientation = LinearLayout.VERTICAL }

        feedbackView = TextView(this).apply {
            textSize = 18f
            setTextColor(Color.parseColor("#FF5B5B"))
            setPadding(0, dp(16), 0, dp(8))
        }

        col.addView(title)
        col.addView(progressView)
        col.addView(stemView)
        col.addView(optionsBox)
        col.addView(feedbackView)
        root.addView(col)
        setContentView(root)
        updateProgress()
    }

    private fun updateProgress() {
        progressView.text = "已连续答对 $correctCount / $requiredStreak 题"
    }

    private fun showNextQuestion() {
        val qs = QuizStore.questions(this)
        if (qs.isEmpty()) {
            Toast.makeText(this, "题库为空，请先导入题库", Toast.LENGTH_LONG).show()
            finish()
            return
        }
        val idx = QuizStore.takeNext(this)
        if (idx < 0 || idx >= qs.size) { finish(); return }
        val q = qs[idx]
        current = q
        render(q)
    }

    private fun render(q: Question) {
        stemView.text = if (q.isJudge) "【判断题】${q.stem}" else "【单选题】${q.stem}"
        feedbackView.text = ""
        updateProgress()
        optionsBox.removeAllViews()

        if (q.isJudge) {
            // q.correct==1 表示答案为“正确”，==0 表示答案为“错误”
            optionsBox.addView(makeOptionButton("正确", q.correct == 1, q))
            optionsBox.addView(makeOptionButton("错误", q.correct == 0, q))
        } else {
            // 选项连内容一起随机打乱，正确答案跟随其文本移动
            val items = ArrayList<Pair<String, Int>>()
            for (i in q.options.indices) items.add(q.options[i] to i) // (文本, 原下标)
            items.shuffle(Random.Default)
            items.forEachIndexed { pos, pair ->
                val letter = ('A' + pos)
                val isCorrect = (pair.second == q.correct)
                optionsBox.addView(
                    makeOptionButton("$letter. ${pair.first}", isCorrect, q)
                )
            }
        }
    }

    private fun makeOptionButton(label: String, isCorrect: Boolean, q: Question): Button {
        val b = Button(this).apply {
            text = label
            textSize = 20f
            isAllCaps = false
            setPadding(dp(16), dp(14), dp(16), dp(14))
            layoutParams = LinearLayout.LayoutParams(
                LinearLayout.LayoutParams.MATCH_PARENT,
                LinearLayout.LayoutParams.WRAP_CONTENT
            ).apply { bottomMargin = dp(10) }
            setBackgroundColor(Color.parseColor("#16243B"))
            setTextColor(Color.WHITE)
        }
        b.setOnClickListener {
            if (isCorrect) {
                correctCount++
                updateProgress()
                if (correctCount >= requiredStreak) {
                    // 达成连对数：放行 + 重新计时
                    feedbackView.setTextColor(Color.parseColor("#2BB673"))
                    feedbackView.text = "✅ 连对 ${requiredStreak} 题，继续看吧！"
                    LockService.scheduleNext(this)
                    Handler(Looper.getMainLooper()).postDelayed({ finish() }, 800)
                } else {
                    // 还需继续连对：追加下一道新题
                    feedbackView.setTextColor(Color.parseColor("#2BB673"))
                    feedbackView.text = "✅ 答对了！再答对 ${requiredStreak - correctCount} 题即可解锁"
                    showNextQuestion()
                }
            } else {
                // 答错：连对清零，追加一道新题，仍锁屏
                correctCount = 0
                updateProgress()
                feedbackView.setTextColor(Color.parseColor("#FF5B5B"))
                val tip = if (q.analysis.isNotBlank())
                    "❌ 答错了，计数清零。${q.analysis}  再来一题"
                else "❌ 答错了，计数清零，再来一题"
                feedbackView.text = tip
                showNextQuestion()
            }
        }
        return b
    }

    override fun onKeyDown(keyCode: Int, event: KeyEvent): Boolean {
        if (keyCode == KeyEvent.KEYCODE_BACK || keyCode == KeyEvent.KEYCODE_HOME) return true
        return super.onKeyDown(keyCode, event)
    }

    override fun onBackPressed() { /* 禁用 */ }

    private fun dp(v: Int): Int = (v * resources.displayMetrics.density).toInt()
}

        col.addView(rowS)

        // 入口A
        col.addView(subhead("【入口A】粘贴导入题库 JSON："))
        pasteInput = EditText(this).apply {
            hint = "把 AI 生成的题库 JSON 粘贴到这里（需含 questions 数组）"
            setTextColor(Color.WHITE)
            setHintTextColor(Color.parseColor("#6B7A93"))
            setBackgroundColor(Color.parseColor("#16243B"))
            minHeight = dp(120)
            gravity = Gravity.TOP or Gravity.START
        }

        col.addView(subhead("【入口B】题库链接自动同步："))

            hint = "填公开可读的 .json 地址，如 https://…/题库.json"

        rowB.addView(btn("保存地址", 1f) { saveUrl() })

            if (QuizStore.hasBank(this)) {
                startActivity(Intent(this, QuizLockActivity::class.java))

                Toast.makeText(this, "还没有题库，先用入口A/B放题", Toast.LENGTH_SHORT).show()
            }

        setContentView(root)

    // ---- 业务动作 ----
    private fun bumpInterval(delta: Int) {
        val v = (QuizStore.getInterval(this) + delta).coerceIn(1, 1440)
        setInterval(v)
    }

    private fun setInterval(v: Int) {
        QuizStore.setInterval(this, v)
        LockService.scheduleNext(this)
        refreshStatus()
    }

    private fun bumpStreak(delta: Int) {
        setStreak((QuizStore.getStreak(this) + delta).coerceIn(1, 10))
    }

    private fun setStreak(v: Int) {
        QuizStore.setStreak(this, v)
        streakLabel.text = v.toString()
        refreshStatus()
    }

    private fun importFromPaste() {
        val text = pasteInput.text.toString()
        if (text.isBlank()) { Toast.makeText(this, "请先粘贴内容", Toast.LENGTH_SHORT).show(); return }
        try {
            val n = QuizParser.parse(text).size
            QuizStore.saveRaw(this, text)
            Toast.makeText(this, "导入成功，共 $n 题", Toast.LENGTH_LONG).show()
            pasteInput.setText("")
            refreshStatus()
        } catch (e: Exception) {
            Toast.makeText(this, "格式有误：${e.message}", Toast.LENGTH_LONG).show()
        }
    }

    private fun saveUrl() {
        QuizStore.setSyncUrl(this, urlInput.text.toString())
        Toast.makeText(this, "同步地址已保存", Toast.LENGTH_SHORT).show()
        refreshStatus()
    }

    private fun doSync() {
        QuizStore.setSyncUrl(this, urlInput.text.toString())
        QuizSync.fetchNow(this, object : QuizSync.Result {
            override fun onDone(ok: Boolean, msg: String) {
                Toast.makeText(this@MainActivity, msg, Toast.LENGTH_LONG).show()
                if (ok) refreshStatus()
            }
        })
    }

    private fun refreshStatus() {
        val count = QuizStore.questions(this).size
        intervalLabel.text = QuizStore.getInterval(this).toString()
        val last = QuizStore.getLastSync(this)
        val lastStr = if (last == 0L) "从未同步"
        else "上次同步 ${android.text.format.DateFormat.getDateFormat(this).format(java.util.Date(last))}"
        statusView.text = "当前题库：$count 题　|　间隔 ${QuizStore.getInterval(this)} 分钟　|　连对 ${QuizStore.getStreak(this)} 题解锁　|　$lastStr"
    }

    // ---- 小工具 ----
    private fun header(t: String) = TextView(this).apply {
        text = t; textSize = 26f; setTypeface(typeface, Typeface.BOLD)
        setTextColor(Color.parseColor("#2BB673"))
    }

    private fun subhead(t: String) = TextView(this).apply {
        text = t; textSize = 17f; setTextColor(Color.WHITE)
        setPadding(0, dp(18), 0, dp(8))
    }

    private fun labelView(t: String, w: Float) = TextView(this).apply {
        text = t; textSize = 22f; setTextColor(Color.WHITE); gravity = Gravity.CENTER
        layoutParams = LinearLayout.LayoutParams(0, dp(48), w)
    }

    private fun btn(t: String, w: Float, onClick: () -> Unit) = Button(this).apply {
        text = t; isAllCaps = false; textSize = 15f
        layoutParams = LinearLayout.LayoutParams(0, dp(48), w)
        setBackgroundColor(Color.parseColor("#16243B")); setTextColor(Color.WHITE)
        setOnClickListener { onClick() }
    }

    private fun btnFull(t: String, onClick: () -> Unit) = Button(this).apply {
        text = t; isAllCaps = false; textSize = 18f
        layoutParams = LinearLayout.LayoutParams(
            LinearLayout.LayoutParams.MATCH_PARENT, dp(54)
        ).apply { topMargin = dp(12) }
        setBackgroundColor(Color.parseColor("#0F8A4F")); setTextColor(Color.WHITE)
        setOnClickListener { onClick() }
    }

    private fun dp(v: Int): Int = (v * resources.displayMetrics.density).toInt()
}

## 功能（已按你的要求实现）
- 电视**开机后先弹一道题**，答对才能看
- 之后**每满 10 分钟**自动弹一道全屏题（间隔可在 App 里调，1~1440 分钟）

# 答题锁 TV · QuizLock TV

> 一个给小米/安卓智能电视的「答题解锁看电视」App。孩子开机后必须先答题、看电视每隔几分钟再弹一次全屏题，**答对才能继续看、答错就追加新题直到答对**。题目由 AI（如 WorkBuddy / DeepSeek / 智谱 GLM 等）生成，App 本身不内置任何 AI、不需要 API Key、不花一分钱。
>
> 同一 WiFi 下，手机浏览器打开电视内置的控制台网页，即可**随时更新题库、调整弹窗规则、手动弹题**——控制权在家长手机里，孩子拿遥控器完全绕不过。

**作者：[Gavin-Space](https://github.com/Gavin-Space)**

---

## ✨ 功能特性

- **开机先答题**：电视开机自动弹一道全屏锁屏题，答对前看不了电视。
- **每 N 分钟弹一题**：观看过程中每隔 10 分钟（间隔可调 1–1440 分钟）自动弹全屏题。
- **全屏锁屏、防绕过**：覆盖电视画面、屏蔽返回键/Home 键、`FLAG_SECURE` 防截屏，无法用遥控器跳过。
- **单选 + 判断**：每次只弹一道题，题型混用。
- **答对解锁，答错追加**：答对则放行并重新计时；答错追加一道新题，直到答对为止。
- **防"闭眼乱点蒙对"**：
  - 默认需**连续答对 3 题**才解锁，答错计数清零重来（乱点蒙过率从单题 25% 降到约 1.6%）；连对数可调 1–10。
  - 单选题**选项每次随机打乱**，杜绝"肌肉记忆点第一个/固定选 A"。
- **手机远程控制（同 WiFi）**：电视内置局域网 HTTP 服务器，手机/平板/电脑浏览器打开即用，安卓和 iPhone 通吃，无需安装任何手机端 App。
- **纯离线、零成本**：不绑定任何大模型或付费接口，题目格式开放，谁来出题都行。

## 🎯 它解决的真实问题

新闻里那种"爸爸魔改电视、儿子看 10 分钟答一题"的效果，普通家庭也能复刻：把"看电视"变成"闯关答题"，且锁屏、计时、放题的开关都在家长手机里，孩子断电重启、按返回键、乱点选项都突破不了。

## 📦 安装

1. 在 Releases / Actions 产物里下载 `app-debug.apk`。
2. 电视：`设置 → 关于 → 连续点"型号/版本号" 7 次` 进入开发者模式，打开 **USB/网络调试**。
3. 电脑装 adb，`adb connect 电视IP:5555`（首次需在电视上点"允许调试"），再 `adb install app-debug.apk`。
4. 首次使用进电视 `设置 → 应用 → 答题锁TV → 权限`，授予 **悬浮窗 + 自启动 + 忽略电池优化**，否则待机时弹窗可能被系统清理。

## 🚀 快速开始

1. 打开 App，主页会显示 **📱 手机控制台地址**，形如 `http://192.168.1.50:8770`。
2. 手机连**和电视同一个 WiFi**，浏览器打开该地址 → 进入控制台。
3. 用 AI 生成题库（见下方提示词模板），把 JSON 粘进控制台"更新题库"框 → 推送。
4. 之后：开机自动弹题、每隔设定分钟弹题、随时手动弹窗，全部生效。

> ⚠️ 手机打不开控制台，多半是路由器开了「AP 隔离 / 访客网络」把设备隔开了，换主 WiFi 即可。

## 📝 题库格式

```json
{
  "questions": [
    {"type": "single", "stem": "下列函数中，在 (0,+∞) 上为减函数的是？",
     "options": ["y = x²", "y = 2x", "y = 1/x", "y = √x"], "answer": 2, "analysis": "y=1/x 在 (0,+∞) 递减"},
    {"type": "judge", "stem": "0 是自然数。", "answer": true, "analysis": "现行教材规定 0∈N"}
  ]
}
```

| 字段 | 说明 |
|---|---|
| `type` | `"single"` 单选 / `"judge"` 判断 |
| `stem` | 题干 |
| `options` | 单选的选项数组（判断题省略或留空） |
| `answer` | 单选填**正确项下标**（从 0 数）；判断填 `true` / `false` |
| `analysis` | 解析（答错时显示给孩子看，可选） |

## 🧩 架构

```
电视 App（Android）
├── MainActivity        管理主页 / 显示手机控制台地址
├── QuizLockActivity    全屏锁屏出题（连对解锁、选项乱序、屏蔽返回键）
├── LockService         前台常驻服务：每 N 分钟闹钟 + 拉起控制服务器
├── BootReceiver        开机广播 → 弹首题 + 启动服务
├── AlarmReceiver       定时闹钟 → 弹窗
├── ControlServer       内置局域网 HTTP 服务器（端口 8770）
│     GET  /            → 手机控制台网页（assets/console.html）
│     GET  /api/status   → 题库数 / 间隔 / 连对 / IP
│     POST /api/bank     → 接收题库 JSON，校验后整库替换
│     POST /api/config   → 改弹窗间隔 / 连对数
│     POST /api/popup    → 立即弹全屏锁屏
├── QuizStore           SharedPreferences 持久化题库 / 配置 / 进度
├── QuizParser          题库 JSON 解析校验
└── QuizSync            备用：从 URL 拉取题库
```

- `minSdk 21`（Android 5.0）兼容绝大多数国行小米电视（MIUI TV / 澎湃 OS 多为 Android 9~11）；`targetSdk 30` 放宽后台弹全屏锁屏限制。
- 控制服务器用纯 JDK `ServerSocket` 实现，不引入第三方 HTTP 库，兼容性最稳。
