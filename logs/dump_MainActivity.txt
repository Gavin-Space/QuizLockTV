  17: /**
  18:  * 管理主页 = 题库入口。
  19:  * 入口A：粘贴导入（把 AI 生成的题库 JSON 粘进来）。
  20:  * 入口B：URL 自动同步（填一个公开 .json 地址，电视定时/手动拉取更新）。
  21:  * 另：设置出题间隔、测试锁屏。
  22:  */
  23: class MainActivity : Activity() {
  24: 
  25:     private lateinit var statusView: TextView
  26:     private lateinit var intervalLabel: TextView
  27:     private lateinit var streakLabel: TextView
  28:     private lateinit var pasteInput: EditText
  29:     private lateinit var urlInput: EditText
  30: 
  31:     override fun onCreate(savedInstanceState: Bundle?) {
  32:         super.onCreate(savedInstanceState)
  33:         buildLayout()
  34:         refreshStatus()
  35:         startServiceAndSchedule()
  36:     }
  37: 
  38:     private fun startServiceAndSchedule() {
  39:         val svc = Intent(this, LockService::class.java)
  40:         runCatching { startForegroundService(svc) }
  41:             .onFailure { runCatching { startService(svc) } }
  42:         LockService.scheduleNext(this)
  43:     }
  44: 
  45:     private fun buildLayout() {
  46:         val root = ScrollView(this).apply {
  47:             setBackgroundColor(Color.parseColor("#0B1220"))
  48:             isFillViewport = true
  49:         }
  50:         val col = LinearLayout(this).apply {
  51:             orientation = LinearLayout.VERTICAL
  52:             setPadding(dp(24), dp(24), dp(24), dp(24))
  53:         }
  54: 
  55:         col.addView(header("答题锁 TV"))
  56: 
  57:         // 手机控制台地址（同一WiFi下手机浏览器打开即可远程控制）
  58:         col.addView(TextView(this).apply {
  59:             text = "📱 手机控制台（同WiFi下手机浏览器打开）：\n" + NetworkInfo.consoleUrl()
  60:             textSize = 16f
  61:             setTextColor(Color.parseColor("#1f6feb"))
  62:             setTypeface(typeface, Typeface.BOLD)
  63:             setPadding(0, dp(4), 0, dp(4))
  64:             setBackgroundColor(Color.parseColor("#0B2A4A"))
  65:             gravity = Gravity.CENTER
  66:             setPadding(dp(12), dp(12), dp(12), dp(12))
  67:         })
  68: 
  69:         statusView = TextView(this).apply {
  70:             textSize = 16f; setTextColor(Color.parseColor("#9FB0CC"))
  71:             setPadding(0, dp(6), 0, dp(16))
  72:         }
  73:         col.addView(statusView)
  74: 
  75:         // 间隔设置
  76:         col.addView(subhead("每看几分钟弹一题："))
  77:         val row = LinearLayout(this).apply { orientation = LinearLayout.HORIZONTAL }
  78:         row.addView(btn("-1", 1f) { bumpInterval(-1) })
  79:         intervalLabel = labelView(QuizStore.getInterval(this).toString(), 1f)
  80:         row.addView(intervalLabel)
  81:         row.addView(btn("+1", 1f) { bumpInterval(1) })
  82:         row.addView(btn("设为10", 1.5f) { setInterval(10) })
  83:         col.addView(row)
  84: 
  85:         // 防乱点：需连续答对几题才解锁
  86:         col.addView(subhead("需连续答对几题才解锁（越大越难乱点）："))
  87:         val rowS = LinearLayout(this).apply { orientation = LinearLayout.HORIZONTAL }
  88:         rowS.addView(btn("-1", 1f) { bumpStreak(-1) })
  89:         streakLabel = labelView(QuizStore.getStreak(this).toString(), 1f)
  90:         rowS.addView(streakLabel)
  91:         rowS.addView(btn("+1", 1f) { bumpStreak(1) })
  92:         rowS.addView(btn("设为3", 1.5f) { setStreak(3) })
  93:         col.addView(rowS)
  94: 
  95:         // 入口A
  96:         col.addView(subhead("【入口A】粘贴导入题库 JSON："))
  97:         pasteInput = EditText(this).apply {
  98:             hint = "把 AI 生成的题库 JSON 粘贴到这里（需含 questions 数组）"
  99:             setTextColor(Color.WHITE)
 100:             setHintTextColor(Color.parseColor("#6B7A93"))
 101:             setBackgroundColor(Color.parseColor("#16243B"))
 102:             minHeight = dp(120)
 103:             gravity = Gravity.TOP or Gravity.START
 104:         }
 105:         col.addView(pasteInput)
 106:         col.addView(btnFull("保存并导入") { importFromPaste() })
 107: 
 108:         // 入口B
 109:         col.addView(subhead("【入口B】题库链接自动同步："))
 110:         urlInput = EditText(this).apply {
 111:             hint = "填公开可读的 .json 地址，如 https://…/题库.json"
 112:             setTextColor(Color.WHITE)
 113:             setHintTextColor(Color.parseColor("#6B7A93"))
 114:             setBackgroundColor(Color.parseColor("#16243B"))
 115:             setText(QuizStore.getSyncUrl(this@MainActivity))
 116:         }
 117:         col.addView(urlInput)
 118:         val rowB = LinearLayout(this).apply { orientation = LinearLayout.HORIZONTAL }
 119:         rowB.addView(btn("保存地址", 1f) { saveUrl() })
 120:         rowB.addView(btn("立即同步", 1f) { doSync() })
 121:         col.addView(rowB)
 122: 
 123:         // 测试
 124:         col.addView(btnFull("测试：立即弹一道题") {
 125:             if (QuizStore.hasBank(this)) {
 126:                 startActivity(Intent(this, QuizLockActivity::class.java))
 127:             } else {
 128:                 Toast.makeText(this, "还没有题库，先用入口A/B放题", Toast.LENGTH_SHORT).show()
 129:             }
 130:         })
 131: 
 132:         root.addView(col)
 133:         setContentView(root)
 134:     }
 135: 
 136:     // ---- 业务动作 ----
 137:     private fun bumpInterval(delta: Int) {
 138:         val v = (QuizStore.getInterval(this) + delta).coerceIn(1, 1440)
 139:         setInterval(v)
 140:     }
 141: 
 142:     private fun setInterval(v: Int) {
 143:         QuizStore.setInterval(this, v)
 144:         LockService.scheduleNext(this)
 145:         refreshStatus()
 146:     }
 147: 
 148:     private fun bumpStreak(delta: Int) {
 149:         setStreak((QuizStore.getStreak(this) + delta).coerceIn(1, 10))
 150:     }
 151: 
 152:     private fun setStreak(v: Int) {
 153:         QuizStore.setStreak(this, v)
 154:         streakLabel.text = v.toString()
 155:         refreshStatus()
 156:     }
 157: 
 158:     private fun importFromPaste() {
 159:         val text = pasteInput.text.toString()
 160:         if (text.isBlank()) { Toast.makeText(this, "请先粘贴内容", Toast.LENGTH_SHORT).show(); return }
 161:         try {
 162:             val n = QuizParser.parse(text).size
 163:             QuizStore.saveRaw(this, text)
 164:             Toast.makeText(this, "导入成功，共 $n 题", Toast.LENGTH_LONG).show()
 165:             pasteInput.setText("")
 166:             refreshStatus()
 167:         } catch (e: Exception) {
 168:             Toast.makeText(this, "格式有误：${e.message}", Toast.LENGTH_LONG).show()
 169:         }
 170:     }
 171: 
 172:     private fun saveUrl() {
 173:         QuizStore.setSyncUrl(this, urlInput.text.toString())
 174:         Toast.makeText(this, "同步地址已保存", Toast.LENGTH_SHORT).show()
 175:         refreshStatus()
 176:     }
 177: 
 178:     private fun doSync() {
 179:         QuizStore.setSyncUrl(this, urlInput.text.toString())
 180:         QuizSync.fetchNow(this, object : QuizSync.Result {
 181:             override fun onDone(ok: Boolean, msg: String) {
 182:                 Toast.makeText(this@MainActivity, msg, Toast.LENGTH_LONG).show()
 183:                 if (ok) refreshStatus()
 184:             }
 185:         })
 186:     }
 187: 
 188:     private fun refreshStatus() {
 189:         val count = QuizStore.questions(this).size
 190:         intervalLabel.text = QuizStore.getInterval(this).toString()
 191:         val last = QuizStore.getLastSync(this)
 192:         val lastStr = if (last == 0L) "从未同步"
 193:         else "上次同步 ${android.text.format.DateFormat.getDateFormat(this).format(java.util.Date(last))}"
 194:         statusView.text = "当前题库：$count 题　|　间隔 ${QuizStore.getInterval(this)} 分钟　|　连对 ${QuizStore.getStreak(this)} 题解锁　|　$lastStr"
 195:     }
 196:

 201:     }
 202:

 206:     }
 207:

 209:         text = t; textSize = 22f; setTextColor(Color.WHITE); gravity = Gravity.CENTER
 210:         layoutParams = LinearLayout.LayoutParams(0, dp(48), w)

 212:

 214:         text = t; isAllCaps = false; textSize = 15f
 215:         layoutParams = LinearLayout.LayoutParams(0, dp(48), w)
 216:         setBackgroundColor(Color.parseColor("#16243B")); setTextColor(Color.WHITE)
 217:         setOnClickListener { onClick() }
 218:     }

 221:         text = t; isAllCaps = false; textSize = 18f
 222:         layoutParams = LinearLayout.LayoutParams(

 225:         setBackgroundColor(Color.parseColor("#0F8A4F")); setTextColor(Color.WHITE)
 226:         setOnClickListener { onClick() }
 227:     }
 228:

 230: }
