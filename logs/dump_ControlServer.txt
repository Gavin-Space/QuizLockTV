  17: /**
  18:  * 内置局域网 HTTP 控制服务器：让同一 WiFi 下的手机/电脑用浏览器控制电视。
  19:  * 路由：
  20:  *   GET  /             -> 手机控制台网页(assets/console.html)
  21:  *   GET  /api/status   -> 当前题库数/间隔/连对/IP 的 JSON
  22:  *   POST /api/bank     -> body 为题库 JSON，校验后整库替换
  23:  *   POST /api/config   -> body: interval=10&streak=3，改弹窗间隔与连对数
  24:  *   POST /api/popup    -> 立即拉起全屏锁屏出题
  25:  * 纯 JDK ServerSocket 实现，不依赖第三方库，targetSdk30 无 cleartext 限制问题。
  26:  */
  27: class ControlServer(private val ctx: Context, private val port: Int = NetworkInfo.PORT) {
  28: 
  29:     @Volatile private var running = false
  30:     private var server: ServerSocket? = null
  31: 
  32:     fun start() {
  33:         if (running) return
  34:         running = true
  35:         thread(isDaemon = true) {
  36:             try {
  37:                 server = ServerSocket().apply {
  38:                     reuseAddress = true
  39:                     bind(InetSocketAddress(port))
  40:                 }
  41:                 while (running) {
  42:                     val socket = try { server?.accept() ?: break } catch (e: Exception) { if (!running) break else continue }
  43:                     thread(isDaemon = true) { handle(socket) }
  44:                 }
  45:             } catch (e: Exception) {
  46:                 // 端口占用等，忽略
  47:             }
  48:         }
  49:     }
  50: 
  51:     fun stop() {
  52:         running = false
  53:         runCatching { server?.close() }
  54:     }
  55: 
  56:     private fun handle(socket: Socket) {
  57:         try {
  58:             socket.use { s ->
  59:                 s.soTimeout = 8000
  60:                 val input = s.getInputStream()
  61:                 val output = s.getOutputStream()
  62:                 val reader = input.reader(StandardCharsets.UTF_8).buffered()
  63: 
  64:                 val requestLine = reader.readLine() ?: return
  65:                 val parts = requestLine.split(" ")
  66:                 if (parts.size < 2) return
  67:                 val method = parts[0]
  68:                 val path = parts[1]
  69: 
  70:                 // 读头部，拿 Content-Length
  71:                 var contentLength = 0
  72:                 while (true) {
  73:                     val line = reader.readLine() ?: break
  74:                     if (line.isEmpty()) break
  75:                     if (line.lowercase().startsWith("content-length:")) {
  76:                         contentLength = line.substringAfter(":").trim().toIntOrNull() ?: 0
  77:                     }
  78:                 }
  79: 
  80:                 val body = if (contentLength > 0) readNChars(reader, contentLength) else ""
  81: 
  82:                 route(method, path, body, output)
  83:             }
  84:         } catch (e: Exception) {
  85:             // 单次请求出错不影响服务
  86:         }
  87:     }
  88: 
  89:     private fun readNChars(reader: BufferedReader, n: Int): String {
  90:         val buf = CharArray(n)
  91:         var read = 0
  92:         while (read < n) {
  93:             val r = reader.read(buf, read, n - read)
  94:             if (r == -1) break
  95:             read += r
  96:         }
  97:         return String(buf, 0, read)
  98:     }
  99: 
 100:     private fun route(method: String, path: String, body: String, out: OutputStream) {
 101:         when {
 102:             method == "GET" && (path == "/" || path.startsWith("/index")) -> {
 103:                 val html = readAsset("console.html")
 104:                 send(out, 200, "text/html; charset=utf-8", html)
 105:             }
 106:             method == "GET" && path.startsWith("/api/status") -> {
 107:                 send(out, 200, "application/json; charset=utf-8", statusJson())
 108:             }
 109:             method == "POST" && path.startsWith("/api/bank") -> {
 110:                 handleBank(body, out)
 111:             }
 112:             method == "POST" && path.startsWith("/api/config") -> {
 113:                 handleConfig(body, out)
 114:             }
 115:             method == "POST" && path.startsWith("/api/popup") -> {
 116:                 handlePopup(out)
 117:             }
 118:             else -> send(out, 404, "text/plain; charset=utf-8", "not found")
 119:         }
 120:     }
 121: 
 122:     private fun statusJson(): String {
 123:         val count = QuizStore.questions(ctx).size
 124:         val interval = QuizStore.getInterval(ctx)
 125:         val streak = QuizStore.getStreak(ctx)
 126:         return """{"count":$count,"interval":$interval,"streak":$streak,"ip":"${NetworkInfo.localIp()}","port":$port}"""
 127:     }
 128: 
 129:     private fun handleBank(body: String, out: OutputStream) {
 130:         try {
 131:             val n = QuizParser.parse(body).size
 132:             QuizStore.saveRaw(ctx, body)
 133:             send(out, 200, "application/json; charset=utf-8", """{"ok":true,"count":$n}""")
 134:         } catch (e: Exception) {
 135:             send(out, 200, "application/json; charset=utf-8", """{"ok":false,"msg":"${jsonEsc(e.message ?: "格式错误")}"}""")
 136:         }
 137:     }
 138: 
 139:     private fun handleConfig(body: String, out: OutputStream) {
 140:         val map = parseForm(body)
 141:         map["interval"]?.toIntOrNull()?.let { QuizStore.setInterval(ctx, it) }
 142:         map["streak"]?.toIntOrNull()?.let { QuizStore.setStreak(ctx, it) }
 143:         LockService.scheduleNext(ctx)
 144:         send(out, 200, "application/json; charset=utf-8", statusJson())
 145:     }
 146: 
 147:     private fun handlePopup(out: OutputStream) {
 148:         val i = Intent(ctx, QuizLockActivity::class.java).addFlags(Intent.FLAG_ACTIVITY_NEW_TASK)
 149:         runCatching { ctx.startActivity(i) }
 150:         send(out, 200, "application/json; charset=utf-8", """{"ok":true}""")
 151:     }
 152: 
 153:     // ---- 工具 ----
 154:     private fun send(out: OutputStream, code: Int, type: String, body: String) {
 155:         val bytes = body.toByteArray(StandardCharsets.UTF_8)
 156:         val status = if (code == 200) "200 OK" else "$code ERR"
 157:         val header = "HTTP/1.1 $status\r\n" +
 158:                 "Content-Type: $type\r\n" +
 159:                 "Content-Length: ${bytes.size}\r\n" +
 160:                 "Cache-Control: no-store\r\n" +
 161:                 "Connection: close\r\n\r\n"
 162:         out.write(header.toByteArray(StandardCharsets.US_ASCII))
 163:         out.write(bytes)
 164:         out.flush()
 165:     }
 166: 
 167:     private fun readAsset(name: String): String {
 168:         return try {
 169:             ctx.assets.open(name).use { it.readBytes().toString(StandardCharsets.UTF_8) }
 170:         } catch (e: Exception) {
 171:             "控制台加载失败：${e.message}"
 172:         }
 173:     }
 174: 
 175:     private fun parseForm(body: String): Map<String, String> {
 176:         val m = HashMap<String, String>()
 177:         body.split("&").forEach { kv ->
 178:             if (kv.contains("=")) {
 179:                 val k = kv.substringBefore("=")
 180:                 val v = kv.substringAfter("=")
 181:                 m[k] = URLDecoder.decode(v, "UTF-8")
 182:             }
 183:         }
 184:         return m
 185:     }
 186: 
 187:     private fun jsonEsc(s: String): String = s.replace("\\", "\\\\").replace("\"", "\\\"").replace("\n", " ").replace("\r", " ")
 188: }
